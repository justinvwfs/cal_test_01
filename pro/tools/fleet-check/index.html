function doGet(e) {
  var companyName = e.parameter.q || '';
  var url = 'https://www.mvdis.gov.tw/m3-emv-mk3/rentalVehicle/query';

  // Step 1: GET to grab session cookies
  var getOpts = {
    method: 'get',
    muteHttpExceptions: true,
    followRedirects: true,
    headers: {
      'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/120'
    }
  };
  var cookies = '';
  try {
    var getResp = UrlFetchApp.fetch(url, getOpts);
    var rawCookies = getResp.getHeaders()['Set-Cookie'];
    if (rawCookies) {
      if (typeof rawCookies === 'string') {
        cookies = rawCookies.split(';')[0];
      } else if (Array.isArray(rawCookies)) {
        cookies = rawCookies.map(function(c){ return c.split(';')[0]; }).join('; ');
      }
    }
  } catch(e) {}

  // Step 2: POST with form data
  var postOpts = {
    method: 'post',
    contentType: 'application/x-www-form-urlencoded',
    payload: 'companyName=' + encodeURIComponent(companyName) + '&pageNum=1&pageSize=100',
    muteHttpExceptions: true,
    followRedirects: true,
    headers: {
      'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/120',
      'Referer': url,
      'Cookie': cookies
    }
  };

  try {
    var resp = UrlFetchApp.fetch(url, postOpts);
    var html = resp.getContentText('UTF-8');
    var result = { ok: true, html: html, keyword: companyName };
    return ContentService
      .createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    var errResult = { ok: false, error: err.toString(), keyword: companyName };
    return ContentService
      .createTextOutput(JSON.stringify(errResult))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
